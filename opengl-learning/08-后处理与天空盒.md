# 第八章：帧缓冲后处理与天空盒

> 目标：从“画到屏幕”升级为“先画到一张纹理，再对纹理做后处理滤镜贴回屏幕”——这是所有“后期特效”的地基；随后给场景戴上**天空盒**（立方体贴图），并实现基于天空盒的简单反射。
>

## 8.1 思路反转：为什么要把画面先画到纹理里

到目前为止我们一直直接画到窗口（默认帧缓冲）。**FBO（Frame Buffer Object，帧缓冲对象）**允许我们“画到一张纹理上”，之后这张纹理可以：

- 当普通贴图贴到屏幕上（**后处理**：模糊、像素化、边缘检测、色差…都在这步用片元着色器做）；
- 作为另一个物体的贴图（镜子/监控屏/汽车后视镜）；
- 实现阴影映射（Shadow Map，光源视角的深度图）。

一句话心智：**画面也是一种纹理，纹理可以被再加工。**

```text
场景渲染 Pass 1：场景 → FBO(纹理A)
     │
     ▼
后处理 Pass 2：把纹理A 贴满屏幕（一个全屏四边形），片元着色器做滤镜 → 屏幕
```

## 8.2 动手做 1：搭一个“画到纹理 → 贴回屏幕”的最小后处理管线

### 新增 `shaders/postprocess.vert`（全屏四边形，不依赖任何模型矩阵）

```glsl
#version 330 core
layout (location = 0) in vec2 aPos;      // 范围 -1..1，正好铺满 NDC
layout (location = 1) in vec2 aTexCoord;
out vec2 vTexCoord;
void main()
{
    gl_Position = vec4(aPos, 0.0, 1.0);
    vTexCoord = aTexCoord;
}
```

### 新增 `shaders/postprocess.frag`

```glsl
#version 330 core
in vec2 vTexCoord;
out vec4 FragColor;
uniform sampler2D uScreenTexture;   // 上一 Pass 画出的场景纹理

void main()
{
    // 第一个滤镜：反色
    vec3 color = texture(uScreenTexture, vTexCoord).rgb;
    FragColor = vec4(1.0 - color, 1.0);
}
```

### 修改 `src/main.cpp`

创建 FBO 与配套纹理（初始化区）：

```cpp
    // 离屏尺寸（可设成窗口大小；本示例固定 800x600）
    const int W = 800, H = 600;

    unsigned int fbo;
    glGenFramebuffers(1, &fbo);
    glBindFramebuffer(GL_FRAMEBUFFER, fbo);

    // 颜色附件：一张空纹理，场景会画进它
    unsigned int colorTex;
    glGenTextures(1, &colorTex);
    glBindTexture(GL_TEXTURE_2D, colorTex);
    glTexImage2D(GL_TEXTURE_2D, 0, GL_RGB, W, H, 0, GL_RGB,
                 GL_UNSIGNED_BYTE, nullptr);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
    glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0,
                           GL_TEXTURE_2D, colorTex, 0);

    // 渲染缓冲对象：给 FBO 配一份深度/模板缓冲（没有它 3D 物体会画错层）
    unsigned int rbo;
    glGenRenderbuffers(1, &rbo);
    glBindRenderbuffer(GL_RENDERBUFFER, rbo);
    glRenderbufferStorage(GL_RENDERBUFFER, GL_DEPTH24_STENCIL8, W, H);
    glFramebufferRenderbuffer(GL_FRAMEBUFFER, GL_DEPTH_STENCIL_ATTACHMENT,
                              GL_RENDERBUFFER, rbo);

    if (glCheckFramebufferStatus(GL_FRAMEBUFFER) != GL_FRAMEBUFFER_COMPLETE)
        std::cerr << "FBO 不完整！" << std::endl;
    glBindFramebuffer(GL_FRAMEBUFFER, 0);   // 绑回默认（屏幕）
```

再准备一个**全屏四边形**的 VAO（两三角形铺满 NDC，顶点只含位置与 uv，位置 -1..1 正好是 NDC 范围，所以不需要 view/projection 就能铺满屏幕）：

```cpp
    float quadVertices[] = {
        // 位置(x,y)   uv
        -1.0f, -1.0f,  0.0f, 0.0f,
         1.0f, -1.0f,  1.0f, 0.0f,
         1.0f,  1.0f,  1.0f, 1.0f,
        -1.0f, -1.0f,  0.0f, 0.0f,
         1.0f,  1.0f,  1.0f, 1.0f,
        -1.0f,  1.0f,  0.0f, 1.0f,
    };
    unsigned int quadVAO, quadVBO;
    glGenVertexArrays(1, &quadVAO);
    glGenBuffers(1, &quadVBO);
    glBindVertexArray(quadVAO);
    glBindBuffer(GL_ARRAY_BUFFER, quadVBO);
    glBufferData(GL_ARRAY_BUFFER, sizeof(quadVertices), quadVertices,
                 GL_STATIC_DRAW);
    glVertexAttribPointer(0, 2, GL_FLOAT, GL_FALSE, 4*sizeof(float), (void*)0);
    glEnableVertexAttribArray(0);
    glVertexAttribPointer(1, 2, GL_FLOAT, GL_FALSE, 4*sizeof(float),
                          (void*)(2*sizeof(float)));
    glEnableVertexAttribArray(1);
```

主循环改成**两个 Pass**：

```cpp
    while (!glfwWindowShouldClose(window))
    {
        /* ---- 键盘/时间更新略（沿用第五章） ---- */

        // ============ Pass 1：正常渲染场景 → FBO ============
        glBindFramebuffer(GL_FRAMEBUFFER, fbo);
        glViewport(0, 0, W, H);
        glClearColor(0.1f, 0.1f, 0.1f, 1.0f);
        glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

        // ……这里原样画你第五章/第六章的受光立方体场景……

        // ============ Pass 2：后处理，贴到屏幕 ============
        glBindFramebuffer(GL_FRAMEBUFFER, 0);      // 画回屏幕
        glViewport(0, 0, 800, 600);
        glClear(GL_COLOR_BUFFER_BIT);              // 屏幕清空

        glDisable(GL_DEPTH_TEST);                  // 全屏四边形不需要深度
        glUseProgram(postProgram);
        glActiveTexture(GL_TEXTURE0);
        glBindTexture(GL_TEXTURE_2D, colorTex);    // 用场景纹理
        glUniform1i(glGetUniformLocation(postProgram, "uScreenTexture"), 0);
        glBindVertexArray(quadVAO);
        glDrawArrays(GL_TRIANGLES, 0, 6);
        glEnable(GL_DEPTH_TEST);

        glfwSwapBuffers(window);
        glfwPollEvents();
    }
```

**预期结果**：立方体场景整体**反色**显示。你已经完成“后处理管线”——以后所有滤镜只是改 `postprocess.frag` 里几行。

## 8.3 滤镜仓库：改片元着色器就能换效果

把 8.2 片元着色器的主函数换成以下任一段，重编译即可看到不同特效：

灰度（经典）：

```glsl
    vec3 c = texture(uScreenTexture, vTexCoord).rgb;
    float g = dot(c, vec3(0.2126, 0.7152, 0.0722));   // 亮度加权和
    FragColor = vec4(vec3(g), 1.0);
```

像素化（低分辨率采样 + 最近邻放大）：

```glsl
    vec2 uv = vTexCoord;
    uv = floor(uv * 80.0) / 80.0;          // 量化成 80x80 的“大像素”
    FragColor = vec4(texture(uScreenTexture, uv).rgb, 1.0);
```

边缘检测（卷积核，一次采样 4 邻域做差分——真正“后处理”的通用形态）：

```glsl
    float off = 1.0 / 800.0;               // 1 个像素对应的 uv 步长
    vec3 c = texture(uScreenTexture, vTexCoord).rgb;
    vec3 l = texture(uScreenTexture, vTexCoord + vec2(-off, 0)).rgb;
    vec3 r = texture(uScreenTexture, vTexCoord + vec2( off, 0)).rgb;
    vec3 d = texture(uScreenTexture, vTexCoord + vec2(0, -off)).rgb;
    vec3 u = texture(uScreenTexture, vTexCoord + vec2(0,  off)).rgb;
    vec3 e = (l + r + d + u) * 0.25 - c;   // 与周围的差
    FragColor = vec4(c + e * 3.0, 1.0);    // 锐化（把细节反差拉大）
```

- **模糊**同理：采样一圈像素取平均，半径越大越糊；
- 各种高级滤镜（泛光、景深、色调映射、暗角）全是这类“多采样 + 加权组合”，配合多个 Pass 叠加——思路一旦打通，剩下的只是工程。

## 8.4 天空盒：把“背景”做成一个永远包着你的立方体

### 8.4.1 立方体贴图（Cubemap）

普通 2D 纹理按 `(u,v)` 采样；**立方体贴图**按方向向量采样——用 6 张图拼成一个“从内部看”的盒子，任意视线方向都能取到“那个方向看到什么”。

- 加载方式：把 6 张面图 `glTexImage2D` 到同一纹理的不同 target（`GL_TEXTURE_CUBE_MAP_POSITIVE_X` 起连续 6 个）；
- 采样：`texture(uCubemap, direction)`，direction 为视线方向；
- 采样器类型：`samplerCube`。

### 8.4.2 最少实现：把天空盒当“背景”画

最省事方案：**后处理 Pass 里采样天空盒**——在 Pass 1 结束后，画一个巨大的立方体（模型矩阵随摄像机位置平移，保证“永远包围你”），或更简单：在片元着色器里按“未命中场景的像素”直接查天空。

工程上标准做法是“一个带 `samplerCube` 的天空盒着色器 + 一个小立方体 + 关闭深度写入”：

```glsl
// 天空盒顶点着色器：把方向传下去
#version 330 core
layout (location = 0) in vec3 aPos;
uniform mat4 uView;
uniform mat4 uProjection;
out vec3 vDir;
void main()
{
    vDir = aPos;
    // 去掉平移（矩阵 3x3 部分），让天空永远“贴”在相机上
    vec4 pos = uProjection * mat4(mat3(uView)) * vec4(aPos, 1.0);
    gl_Position = pos.xyww;      // w 拉平，让它永远在远裁剪面
}
```

```glsl
// 天空盒片元着色器：按方向取色
#version 330 core
in vec3 vDir;
out vec4 FragColor;
uniform samplerCube uSkybox;
void main()
{
    FragColor = texture(uSkybox, vDir);
}
```

绘制顺序注意：**天空盒最先画**，同时 `glDepthMask(GL_FALSE)`（不写深度，防止它挡住场景物体），画完恢复。

## 8.5 进阶彩蛋：用天空盒做“假反射”

镜面物体不需要真的光线追踪——**反射 = 视线关于法线的镜像方向，去天空盒采样**。在物体片元着色器里：

```glsl
uniform samplerCube uSkybox;
uniform vec3 uViewPos;
// ...
vec3 viewDir = normalize(vWorldPos - uViewPos);   // 指向相机的视线
vec3 reflectDir = reflect(viewDir, normalize(vNormal));
vec3 envColor = texture(uSkybox, reflectDir).rgb; // “假”倒影
FragColor = vec4(envColor * 0.5 + objectColor * 0.5, 1.0);
```

- 这属于**环境映射**：用“周围环境贴图”冒充全局光照，成本极低、观感提升巨大；
- 后续若想真正做基于物理的反射（PBR）、阴影、全局光照，路线图见 README 的“进阶二”规划。

## 8.6 实战与验收

1. **滤镜叠加**：把 8.3 的灰度/像素化/边缘检测各实现一遍，再试着“灰度 + 像素化”叠加（思路：灰度结果再喂给像素化分支）；
2. **FBO 尺寸与窗口联动**：在 `FramebufferSizeCallback` 里同步重建 FBO 纹理与 rbo（提示：删旧纹理 → 重新 `glTexImage2D` → 重新挂附件），窗口缩放后画面不再被拉伸/裁剪；
3. **MSAA + 后处理**：FBO 若想抗锯齿，需要把颜色附件建成多重采样纹理（`glTexImage2DMultisample`），最后 `glBlitFramebuffer` 拷贝回普通纹理再后处理——自行查阅 GL 文档完成；
4. **验收标准**：能画出“Pass1 场景→纹理→Pass2 滤镜→屏幕”流程图；能说出 FBO 为什么需要配 depth/stencil；能解释天空盒为什么必须最后（或最前）画、为什么要把深度写入关掉。

## 本章小结

- FBO = 画到纹理：场景 → 离屏颜色纹理（+ 深度模板），后处理全屏四边形 = 滤镜舞台；
- 滤镜本质是“邻域采样 + 加权组合”，灰度/模糊/锐化/边缘检测一脉相承；
- 立方体贴图 = 方向 → 颜色，天空盒与反射（环境映射）都靠它；
- 至此进阶收官：你已掌握“场景 + 光照 + 后处理 + 天空环境”的完整现代渲染管线骨架。
