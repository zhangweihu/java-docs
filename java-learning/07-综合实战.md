2. 删除图书（按编号）

```java
// ===== Book.java 图书实体类 =====
import java.io.Serializable;

public class Book implements Serializable {
    private static final long serialVersionUID = 1L;

    private String id;        // 编号
    private String title;     // 书名
    private String author;    // 作者
    private double price;     // 价格

    public Book() {}

    public Book(String id, String title, String author, double price, int stock) {
        this.id = id;
        this.title = title;
        this.author = author;
        this.price = price;
        this.stock = stock;
    }

    public String getId() { return id; }
    public void setId(String id) { this.id = id; }
    public String getTitle() { return title; }
    public void setTitle(String title) { this.title = title; }
    public String getAuthor() { return author; }
    public void setAuthor(String author) { this.author = author; }
    public double getPrice() { return price; }
    public void setPrice(double price) { this.price = price; }
    public int getStock() { return stock; }
    public void setStock(int stock) { this.stock = stock; }

    @Override
    public String toString() {
        return String.format("编号:%-6s 书名:%-15s 作者:%-8s 价格:%-8.1f 库存:%d",
                id, title, author, price, stock);
    }
}
```

```java
// ===== BookManager.java 图书管理逻辑 =====
import java.io.*;
import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

public class BookManager {
    // 存储图书的集合
    private final List<Book> books = new ArrayList<>();
    // 数据文件路径
    private static final String DATA_FILE = "d:/project/java/docs/java-learning/library.dat";

    public BookManager() {
        loadFromFile();   // 构造时加载已有数据
    }

    // 添加图书
    public boolean addBook(Book book) {
        // 编号不能重复
        for (Book b : books) {
            if (b.getId().equals(book.getId())) {
                System.out.println("图书编号已存在：" + book.getId());
                return false;
            }
        }
        books.add(book);
        saveToFile();
        System.out.println("添加成功：" + book);
        return true;
    }

    // 按编号删除
    public boolean deleteBook(String id) {
        boolean removed = books.removeIf(b -> b.getId().equals(id));
        if (removed) {
            saveToFile();
            System.out.println("删除成功");
        } else {
            System.out.println("未找到编号为 " + id + " 的图书");
        }
        return removed;
    }

    // 按书名模糊查找（使用 Stream）
    public List<Book> searchByTitle(String keyword) {
        return books.stream()
                .filter(b -> b.getTitle().contains(keyword))
                .collect(Collectors.toList());
    }

    // 修改库存
    public boolean updateStock(String id, int newStock) {
        for (Book b : books) {
            if (b.getId().equals(id)) {
                if (newStock < 0) {
                    System.out.println("库存不能为负数");
                    return false;
                }
                b.setStock(newStock);
                saveToFile();
                System.out.println("修改成功：" + b);
                return true;
            }
        }
        System.out.println("未找到编号为 " + id + " 的图书");
        return false;
    }

    // 显示所有图书
    public void showAll() {
        if (books.isEmpty()) {
            System.out.println("库存为空，暂无图书");
            return;
        }
        System.out.println("================ 图书列表 ================");
        for (Book b : books) {
            System.out.println(b);
        }
        System.out.println("共 " + books.size() + " 本图书");
    }

    // 保存到文件（序列化）
    private void saveToFile() {
        try (ObjectOutputStream oos = new ObjectOutputStream(new FileOutputStream(DATA_FILE))) {
            oos.writeObject(books);
        } catch (IOException e) {
            System.out.println("保存数据失败：" + e.getMessage());
        }
    }

    // 从文件加载（反序列化）
    @SuppressWarnings("unchecked")
    private void loadFromFile() {
        File file = new File(DATA_FILE);
        if (!file.exists()) {
            return;   // 首次运行没有文件
        }
        try (ObjectInputStream ois = new ObjectInputStream(new FileInputStream(file))) {
            List<Book> loaded = (List<Book>) ois.readObject();
            books.addAll(loaded);
            System.out.println("已加载 " + loaded.size() + " 本图书数据");
        } catch (IOException | ClassNotFoundException e) {
            System.out.println("加载数据失败：" + e.getMessage());
        }
    }
}
```

```java
// ===== LibraryApp.java 主程序 =====
import java.util.List;
import java.util.Scanner;

public class LibraryApp {
    public static void main(String[] args) {
        BookManager manager = new BookManager();
        Scanner sc = new Scanner(System.in);

        while (true) {
            System.out.println();
            System.out.println("========= 图书管理系统 =========");
            System.out.println("1. 添加图书");
            System.out.println("2. 删除图书");
            System.out.println("3. 查找图书");
            System.out.println("4. 修改库存");
            System.out.println("5. 显示所有图书");
            System.out.println("0. 退出系统");
            System.out.print("请选择操作：");

            int choice;
            try {
                choice = Integer.parseInt(sc.nextLine());
            } catch (NumberFormatException e) {
                System.out.println("请输入数字！");
                continue;
            }

            switch (choice) {
                case 1:  // 添加
                    System.out.print("编号：");   String id = sc.nextLine();
                    System.out.print("书名：");   String title = sc.nextLine();
                    System.out.print("作者：");   String author = sc.nextLine();
                    System.out.print("价格：");   double price = Double.parseDouble(sc.nextLine());
                    System.out.print("库存：");   int stock = Integer.parseInt(sc.nextLine());
                    manager.addBook(new Book(id, title, author, price, stock));
                    break;
                case 2:  // 删除
                    System.out.print("输入要删除的编号：");
                    manager.deleteBook(sc.nextLine());
                    break;
                case 3:  // 查找
                    System.out.print("输入书名关键字：");
                    List<Book> result = manager.searchByTitle(sc.nextLine());
                    if (result.isEmpty()) {
                        System.out.println("没有找到相关图书");
                    } else {
                        result.forEach(System.out::println);
                    }
                    break;
                case 4:  // 修改库存
                    System.out.print("输入编号："); String bid = sc.nextLine();
                    System.out.print("输入新库存："); int newStock = Integer.parseInt(sc.nextLine());
                    manager.updateStock(bid, newStock);
                    break;
                case 5:  // 显示
                    manager.showAll();
                    break;
                case 0:  // 退出
                    System.out.println("感谢使用，再见！");
                    sc.close();
                    return;
                default:
                    System.out.println("无效选项，请重新选择");
            }
        }
    }
}
```

```java
// ===== dao/StudentDao.java 数据访问层 =====
package dao;

import entity.Student;
import java.sql.*;
import java.util.ArrayList;
import java.util.List;

public class StudentDao {
    // JDBC 连接配置
    private static final String URL = "jdbc:mysql://localhost:3306/java_demo"
            + "?useUnicode=true&characterEncoding=utf8&useSSL=false&serverTimezone=Asia/Shanghai";
    private static final String USER = "root";
    private static final String PASSWORD = "123456";

    private Connection getConnection() throws SQLException {
        return DriverManager.getConnection(URL, USER, PASSWORD);
    }

    // 查询所有学生
    public List<Student> findAll() {
        List<Student> list = new ArrayList<>();
        String sql = "SELECT * FROM student ORDER BY id";
        try (Connection conn = getConnection();
             PreparedStatement ps = conn.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
            while (rs.next()) {
                list.add(mapRow(rs));
            }
        } catch (SQLException e) {
            e.printStackTrace();
        }
        return list;
    }

    // 新增学生
    public boolean add(Student s) {
        String sql = "INSERT INTO student(name, age, score) VALUES(?, ?, ?)";
        try (Connection conn = getConnection();
             PreparedStatement ps = conn.prepareStatement(sql)) {
            ps.setString(1, s.getName());
            ps.setInt(2, s.getAge());
            ps.setDouble(3, s.getScore());
            return ps.executeUpdate() > 0;
        } catch (SQLException e) {
            e.printStackTrace();
            return false;
        }
    }

    // 修改学生
    public boolean update(Student s) {
        String sql = "UPDATE student SET name=?, age=?, score=? WHERE id=?";
        try (Connection conn = getConnection();
             PreparedStatement ps = conn.prepareStatement(sql)) {
            ps.setString(1, s.getName());
            ps.setInt(2, s.getAge());
            ps.setDouble(3, s.getScore());
            ps.setInt(4, s.getId());
            return ps.executeUpdate() > 0;
        } catch (SQLException e) {
            e.printStackTrace();
            return false;
        }
    }

    // 删除学生
    public boolean delete(int id) {
        String sql = "DELETE FROM student WHERE id=?";
        try (Connection conn = getConnection();
             PreparedStatement ps = conn.prepareStatement(sql)) {
            ps.setInt(1, id);
            return ps.executeUpdate() > 0;
        } catch (SQLException e) {
            e.printStackTrace();
            return false;
        }
    }

    // 行映射工具方法
    private Student mapRow(ResultSet rs) throws SQLException {
        Student s = new Student();
        s.setId(rs.getInt("id"));
        s.setName(rs.getString("name"));
        s.setAge(rs.getInt("age"));
        s.setScore(rs.getDouble("score"));
        return s;
    }
}
```

```java
// ===== service/StudentService.java 业务层 =====
package service;

import dao.StudentDao;
import entity.Student;
import java.util.List;

public class StudentService {
    private final StudentDao dao = new StudentDao();

    // 业务校验 + 调用 DAO
    public boolean addStudent(Student s) {
        if (s.getName() == null || s.getName().trim().isEmpty()) {
            System.out.println("姓名不能为空");
            return false;
        }
        if (s.getScore() < 0 || s.getScore() > 100) {
            System.out.println("成绩必须在 0~100 之间");
            return false;
        }
        return dao.add(s);
    }

    public boolean updateStudent(Student s) {
        return dao.update(s);
    }

    public boolean deleteStudent(int id) {
        return dao.delete(id);
    }

    public List<Student> getAllStudents() {
        return dao.findAll();
    }

    // 统计：平均分
    public double getAverageScore() {
        List<Student> all = dao.findAll();
        return all.stream().mapToDouble(Student::getScore).average().orElse(0);
    }

    // 统计：最高分
    public double getMaxScore() {
        List<Student> all = dao.findAll();
        return all.stream().mapToDouble(Student::getScore).max().orElse(0);
    }

    // 统计：及格率（>= 60）
    public double getPassRate() {
        List<Student> all = dao.findAll();
        if (all.isEmpty()) return 0;
        long pass = all.stream().filter(s -> s.getScore() >= 60).count();
        return (double) pass / all.size() * 100;
    }

    // 展示统计数据
    public void showStatistics() {
        List<Student> all = dao.findAll();
        System.out.println("========= 成绩统计 =========");
        System.out.println("学生总数：" + all.size());
        System.out.printf("平均分：%.1f%n", getAverageScore());
        System.out.printf("最高分：%.1f%n", getMaxScore());
        System.out.printf("及格率：%.1f%%%n", getPassRate());
    }
}
```

```java
package app;

import entity.Student;
```
